# 命令行直接生成
有时候需要生成大量的不同形状的SRAM的不同view，GUI操作不方便，可以使用命令行的形式。
下面的命令行与上面的GUI操作是同一效果：
```bash
/data/data_eda2/PDK_Tech/TSMC_22NM_RF_ULL/IP/Memory_Compiler/rf_sp_hde_svt_mvt/r0p0/bin/rf_sp_hde_svt_mvt \
verilog \
-name_case lower \
-mvt BASE \
-bus_notation on \
-ser none \
-site_def off \
-check_instname on \
-frequency 1000 \
-bmux off \
-diodes on \
-activity_factor 50 \
-words 32 \
-drive 6 \
-power_type otc \
-bits 128 \
-instname rf_sp_hde_32_128 \
-retention on \
-libertyviewstyle nldm \
-write_mask off \
-atf off \
-left_bus_delim "[" \
-pwr_gnd_rename vddpe:VDDPE,vddce:VDDCE,vsse:VSSE \
-right_bus_delim "]" \
-redundancy off \
-libname rf_sp_hde \
-write_thru off \
-cust_comment "This is a memory instance" \
-mux 2 \
-top_layer m5-m10 \
-power_gating off \
-back_biasing off \
-ema on \
-vmin_assist on \
-corners ffg_cbestt_0p88v_0p88v_0c,ffg_cbestt_0p88v_0p88v_125c,ffg_cbestt_0p88v_0p88v_m40c,ssg_cworstt_0p72v_0p72v_0c,ssg_cworstt_0p72v_0p72v_125c,ssg_cworstt_0p72v_0p72v_m40c,tt_typical_0p80v_0p80v_25c,tt_typical_0p80v_0p80v_85c

```
